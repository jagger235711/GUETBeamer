\section{Motivation}
\begin{frame}{Why use \LaTeX{} for articles}
    \begin{itemize}
        \item High-quality typesetting (math, tables, figures)
        \item Stable cross-references and bibliographies
        \item Reproducible workflows (scripts, CI)
        \item Works well with AI-assisted writing tools for drafting and polishing
    \end{itemize}
\end{frame}

\section{Environment Setup}
\begin{frame}{Recommended tools}
    \begin{itemize}
        \item Distribution: TinyTeX or TeX Live
        \item Engines: XeLaTeX or LuaLaTeX (better font support)
        \item Build: latexmk (automates multiple runs)
        \item Editor: VSCode + LaTeX Workshop, or Overleaf (cloud)
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Common package installation}
    Typical install command (TeX Live / TinyTeX):
    \begin{lstlisting}
tlmgr install ctex biblatex algorithm2e
             listings xstring biber latexindent
\end{lstlisting}
    Build example:
    \begin{lstlisting}
latexmk -xelatex main.tex
biber main
latexmk -xelatex main.tex
\end{lstlisting}
\end{frame}

\section{Document Structure}
\begin{frame}{Recommended article structure}
    \begin{enumerate}
        \item Abstract
        \item Introduction
        \item Related Work
        \item Method / Theory
        \item Experimental Design
        \item Results \& Analysis
        \item Discussion
        \item Conclusion
        \item References \& Appendix
    \end{enumerate}
\end{frame}

\begin{frame}{Practical sectioning tips}
    \begin{itemize}
        \item Draft an outline first
        \item Keep one main idea per section
        \item Use labels (\texttt{\textbackslash label\{...\}}) and \texttt{\textbackslash ref} for cross-references
        \item Put long code and large tables in appendix or external repo
    \end{itemize}
\end{frame}

\section{Math and Algorithms}
\begin{frame}{Math typesetting best practices}
    \begin{itemize}
        \item Use \texttt{amsmath} environments: \texttt{align}, \texttt{gather}
        \item Number only referenced equations
        \item Define macros for repeated notation (vectors, operators)
    \end{itemize}
\end{frame}

\begin{frame}{Algorithm example}
    \begin{algorithm}[H]
        \caption{Example optimization pseudocode}
        % 注意：algorithm2e用\KwIn和\KwOut，无需algorithmic环境
        \KwIn{initial parameter $x_0$, tolerance $\epsilon$}
        \KwOut{optimized parameter $x^*$}
        initialize $B_0 \leftarrow I$\;
        \While{$\|\nabla f(x_t)\|>\epsilon$}{
            $p_t \leftarrow -B_t \nabla f(x_t)$\;
            perform line search to get $\alpha_t$\;
            $x_{t+1} \leftarrow x_t + \alpha_t p_t$\;
            $s_t \leftarrow x_{t+1}-x_t$, \quad $y_t \leftarrow \nabla f(x_{t+1})-\nabla f(x_t)$\;
            update $B_{t+1}$ by BFGS formula\;
        }
    \end{algorithm}
\end{frame}

\section{Code and Reproducibility}
\begin{frame}{Code management}
    \begin{itemize}
        \item Keep code in a \texttt{code/} directory; reference or link in the paper
        \item For short snippets use \texttt{listings} inline; avoid pasting very long scripts
        \item Provide a requirements file and clear reproduction steps
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Code snippet example}
    \begin{lstlisting}
def gradient_step(w, grad, lr):
    return w - lr * grad

for epoch in range(epochs):
    grad = compute_gradient(w)
    w = gradient_step(w, grad, lr)
\end{lstlisting}
\end{frame}

\section{Bibliography and Citations}
\begin{frame}[fragile]{Using \texttt{biblatex} + \texttt{biber}}
    \begin{itemize}
        \item Preamble:
              \begin{itemize}
                  \item \verb|\usepackage[backend=biber,style=authoryear]{biblatex}|
                  \item \verb|\addbibresource{references.bib}|
              \end{itemize}
        \item Compile process: \texttt{xelatex → biber → xelatex ×2}
        \item If \texttt{biber} missing: install and add to PATH
    \end{itemize}
\end{frame}


\begin{frame}{Fallback: manual references}
    If automated tools unavailable, list references manually in an appendix.
    Prefer \texttt{.bib} management for research papers.
\end{frame}

\section{Automation and Tools}
\begin{frame}{Useful automation}
    \begin{itemize}
        \item \textbf{latexmk} – automated compilation
        \item \textbf{biber} – bibliography backend
        \item \textbf{latexindent} – code/tex formatter
        \item CI (GitHub Actions) – reproducible builds
    \end{itemize}
\end{frame}

\begin{frame}{Common environment issues}
    \begin{itemize}
        \item \texttt{biber: command not found} → install biber and add to PATH
        \item \texttt{latexindent not found} → install and configure editor path
        \item Missing \texttt{.sty} files → \texttt{tlmgr install <pkg>}
    \end{itemize}
\end{frame}

\section{Workflow Example}
\begin{frame}{Recommended workflow}
    \begin{enumerate}
        \item Organize content into \texttt{sections/} files
        \item Keep figures in \texttt{figures/}, code in \texttt{code/}
        \item Use a build script or Makefile
        \item Push to repo, use CI to verify build
    \end{enumerate}
\end{frame}

\begin{frame}[fragile]{Example build script}
    \begin{lstlisting}
#!/bin/bash
set -e
latexmk -xelatex -interaction=nonstopmode main.tex
biber main || true
latexmk -xelatex -interaction=nonstopmode main.tex
latexmk -c
\end{lstlisting}
\end{frame}

\section{Checklist and Best Practices}
\begin{frame}{Pre-submission checklist}
    \begin{itemize}
        \item All references compile and appear correctly
        \item Figures have captions and are referenced
        \item Tables use \texttt{booktabs}, no vertical rules
        \item Code and data are accessible or archived
        \item Spelling and grammar checked
    \end{itemize}
\end{frame}

\begin{frame}{Final tips}
    \begin{itemize}
        \item Start from a minimal working example
        \item Use macros for repeated notation
        \item Script builds; avoid manual multiple-step builds
        \item Keep the project folder organized
    \end{itemize}
\end{frame}
