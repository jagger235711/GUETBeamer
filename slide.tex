%==================== 1. 动机与问题定义 ====================%
\section{研究动机与问题定义}

\begin{frame}{研究动机}
  \begin{itemize}
    \item 序列建模任务（如机器翻译）通常基于 RNN 或 CNN。
    \item 但 RNN 存在“顺序依赖”问题，难以并行，训练慢。
    \item CNN 可并行但难捕获长距离依赖。
    \item 因此，论文提出：能否完全依靠 \textbf{Attention} 替代循环与卷积？
  \end{itemize}
  \vspace{0.3cm}
  \textbf{讲述建议：} 从机器翻译的痛点入手，引出“Attention-only”这一大胆想法。
\end{frame}

\begin{frame}{问题定义}
  \begin{itemize}
    \item 任务：序列到序列（Seq2Seq）建模。
    \item 目标：设计一个\textbf{更快、更并行、捕捉长依赖}的新架构。
    \item 关键问题：
      \begin{itemize}
        \item 如何在不使用RNN/CNN的前提下建模序列关系？
        \item 如何保持信息的顺序感？
      \end{itemize}
  \end{itemize}
\end{frame}

%==================== 2. 相关工作 ====================%
\section{相关工作}

\begin{frame}{相关工作简述}
  \begin{itemize}
    \item RNN：LSTM、GRU —— 解决梯度消失，但串行计算限制并行。
    \item CNN：ConvS2S、ByteNet —— 并行性好，但捕获远程依赖效率低。
    \item Self-Attention —— 在阅读理解、摘要生成中已有应用，但未完全替代RNN。
  \end{itemize}
  \vspace{0.4cm}
  \textbf{核心突破：} Transformer 是首个完全依赖自注意力（Self-Attention）的 Seq2Seq 模型。
\end{frame}

%==================== 3. 方法 ====================%
\section{方法 Methodology}

\begin{frame}{模型整体框架}
  \begin{center}
    \includegraphics[width=0.2\linewidth]{framework.png}
  \end{center}
  \begin{itemize}
    \item 编码器-解码器结构，各包含6层堆叠。
    \item 核心模块：多头自注意力（Multi-Head Self-Attention）。
    \item 通过残差连接 + LayerNorm 稳定训练。
  \end{itemize}
  \textbf{讲述建议：} 用图带讲，强调“并行性 + 层次结构”。
\end{frame}

\begin{frame}{Scaled Dot-Product Attention}
  \begin{itemize}
    \item 输入：Query、Key、Value 三组向量。
    \item 计算：
    \[
    \text{Attention}(Q, K, V) = \text{softmax}\left(\frac{QK^T}{\sqrt{d_k}}\right)V
    \]
    \item 实质：通过相关性加权求和。
    \item 优点：矩阵乘法实现，支持并行，计算高效。
  \end{itemize}
\end{frame}

\begin{frame}{Multi-Head Attention}
  \begin{center}
    \includegraphics[width=0.4\linewidth]{multi_head_attention.png}
  \end{center}
  \begin{itemize}
    \item 多头注意力允许模型关注不同子空间的信息。
    \item 每个“头”独立学习不同的依赖关系。
    \item 八个头并行后再拼接，得到最终输出。
  \end{itemize}
  \vspace{0.4cm}
  \textbf{直观解释：} 每个头像一个“专家”，关注句子中的不同语义关系。
\end{frame}

\begin{frame}{位置编码 Positional Encoding}
  \begin{itemize}
    \item 因为模型不含RNN/CNN，所以需显式加入位置信息。
    \item 采用\textbf{正弦-余弦函数}编码：
    \[
    PE(pos,2i) = \sin\left(\frac{pos}{10000^{2i/d_{model}}}\right)
    \]
    \[
    PE(pos,2i+1) = \cos\left(\frac{pos}{10000^{2i/d_{model}}}\right)
    \]
    \item 能让模型学会按相对位置进行注意。
  \end{itemize}
\end{frame}

%==================== 4. 实验结果 ====================%
\section{实验 Experiments}

\begin{frame}{实验设置}
  \begin{itemize}
    \item 数据集：WMT 2014 英-德（4.5M句对），英-法（36M句对）。
    \item 硬件：8 $\times$ NVIDIA P100 GPU。
    \item 优化器：Adam，学习率调度带 Warm-up（4000步）。
    \item 正则化：Dropout=0.1，Label Smoothing=0.1。
  \end{itemize}
\end{frame}

\begin{frame}{机器翻译结果}
  \begin{center}
    \includegraphics[width=0.6\linewidth]{results.png}
  \end{center}
  \begin{itemize}
    \item EN-DE: BLEU = 28.4（+2.0 优于 SOTA）
    \item EN-FR: BLEU = 41.8（单模型新纪录）
    \item 训练仅需 3.5 天，比 GNMT 节省 75\% 计算量。
  \end{itemize}
  \vspace{0.3cm}
  \textbf{讲述建议：} 强调“性能 + 效率”双突破。
\end{frame}

\begin{frame}{可视化与可解释性}
  \begin{center}
    \includegraphics[width=0.95\linewidth]{attention_vis.png}
  \end{center}
  \vspace{0.2cm}
  \textbf{发现：} 不同注意力头能自动学习语法结构、共指关系等语义模式。
\end{frame}

%==================== 5. 总结 ====================%
\section{总结 Conclusion}

\begin{frame}{总结与展望}
  \begin{itemize}
    \item 提出第一个完全基于注意力的序列转换模型 Transformer。
    \item 实现更强性能、更高并行度、更低训练成本。
    \item 后续方向：
      \begin{itemize}
        \item 将注意力机制扩展到图像、语音、多模态任务。
        \item 优化局部注意力处理长序列。
      \end{itemize}
  \end{itemize}
  \vspace{0.3cm}
  \textbf{一句话总结：} \textit{“Attention replaces recurrence and convolution — and works even better.”}
\end{frame}
