\section{问题定义}
\begin{frame}{核心问题与难点}  
    \begin{block}{核心问题}
        如何有效融合智能制造中的多源异构数据（图像、传感器、文本）以提升实时决策精度？
    \end{block}
    \vspace{0.3cm}
    \textbf{主要难点：}
    \begin{itemize}
        \item 模态间数据格式差异显著：
            \begin{itemize}
                \item 图像数据为像素矩阵
                \item 传感器数据是时序数值
                \item 文本数据是非结构化语义
            \end{itemize}
        \item 跨模态语义对齐困难：
            \begin{itemize}
                \item 传统方法难以建立不同模态间的语义关联
                \item 导致信息孤立，无法形成决策闭环
            \end{itemize}
    \end{itemize}
\end{frame}

\section{研究方法}
\begin{frame}{方法创新}
    % 1. 现有方法局限性：用灰色背景区块区分，突出“问题”属性
    \begin{block}{现有方法局限性}
        \begin{itemize}
            \item 单模态分析局限：仅针对单一数据类型（如纯图像或纯文本处理），无法捕捉跨模态关联信息；
            \item 表征方法缺陷：缺乏动态 tokenization 机制，且未形成统一的多模态表征框架。
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}
    % 2. 本文创新点：用带颜色的强调区块突出，与“局限”形成视觉对比
    \begin{block}{\textcolor{blue}{本文核心创新点}}
        \begin{enumerate} % 用有序列表，明确创新点优先级和数量
            \item \textbf{统一多模态表征}：将图像（像素矩阵）、传感器数据（时序数值）、生产记录（文本）映射到共享语义空间，消除模态差异；
            \item \textbf{动态语义 Tokenization}：自适应调整不同模态的输入编码方式（如图像分块 Token、传感器时序 Token）；
            \item \textbf{跨模态对齐策略}：引入模态间注意力机制，强化图像-文本、传感器-生产记录的语义关联；
            \item \textbf{两阶段训练范式}：先基于大规模工业数据集预训练，再针对具体决策任务（如故障诊断）微调；
            \item \textbf{改进型 Transformer 模型}：同步生成图像特征与文本描述，将决策响应速度提升约30\%。
        \end{enumerate}
    \end{block}
\end{frame}

\section{实验验证}
\begin{frame}{实验结果}
    % 1. 实验设置：用基础区块明确前提，结构清晰
    \begin{block}{实验设置}
        \begin{itemize}
            \item \textbf{数据集}：工业多模态专用数据集（含10万+图像-文本对、500万条传感器时序序列）；
            \item \textbf{基线模型}：CNN+RNN混合模型、单模态LLM（如GPT-2工业微调版）、主流多模态模型（ViLBERT、FLAVA）；
            \item \textbf{评价指标}：图像-文本双向检索准确率、视觉问答（VQA）F1分数、决策响应延迟（单位：ms）。
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}
    % 2. 关键结果：用红色强调区块突出核心数据，视觉冲击力强
    \begin{block}{\textcolor{red}{关键实验结果}}
        \begin{itemize}
            \item \textbf{图像-文本检索准确率}：本文模型达89.5\%，较最优基线（ViLBERT 77.2\%）提升\textbf{12.3\%}；
            \item \textbf{VQA F1分数}：本文模型达82.1\%，较基线（FLAVA 72.3\%）提高\textbf{9.8\%}；
            \item \textbf{决策延迟}：因跨模态并行生成机制，较传统串行模型降低\textbf{30\%}（从280ms降至196ms）；
            \item \textbf{消融实验}：动态语义Tokenization模块贡献最大，移除后性能下降7.1\%，验证其核心作用。
        \end{itemize}
    \end{block}
\end{frame}
