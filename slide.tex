\section{研究背景与目标}
\begin{frame}{研究背景与目标}
  \begin{itemize}
    \item 梯度下降（Gradient Descent, GD）是最基础的优化算法；
    \item 拟牛顿法（BFGS）是一种利用 Hessian 近似的二阶方法；
    \item 实验目标：
          \begin{itemize}
            \item 动手实现梯度下降算法；
            \item 实现拟牛顿法；
            \item 在同一线性模型上比较两者的性能差异。
          \end{itemize}
  \end{itemize}

  \vspace{0.5cm}
  \textbf{核心问题：}谁在速度与精度上更具优势？
\end{frame}

%--------------------------------------------------
\section{实验设计与流程}
\begin{frame}{实验设计与流程}
  \begin{enumerate}
    \item \textbf{数据生成：} 设线性模型
          \[
            y = w^\top x + b + \epsilon, \quad \epsilon \sim \mathcal{N}(0, \sigma^2)
          \]
    \item \textbf{梯度下降实现：}
          \begin{itemize}
            \item 手动计算梯度；
            \item 使用固定学习率更新参数。
          \end{itemize}
    \item \textbf{拟牛顿法实现（BFGS）：}
          \begin{itemize}
            \item 使用 \texttt{scipy.optimize.minimize}；
            \item 自动估计 Hessian 近似矩阵。
          \end{itemize}
  \end{enumerate}
\end{frame}

  \begin{itemize}

    \item GD收敛更快、更稳定；
  \end{itemize}

%--------------------------------------------------
\section{问题与原因分析}
\begin{frame}{问题与原因分析}
  \textbf{主要问题：}
  \begin{enumerate}
    \item \textbf{Loss下降太快}：
          \begin{itemize}
            \item 问题过于简单；
            \item 噪声过小或超参数设置过优；
          \end{itemize}

    \item \textbf{BFGS性能比GD差，与预期不符}：
          \begin{itemize}
            \item 原因：线性模型过于简单，用不上Hessian；
            \item 使用非线性模型。
          \end{itemize}

  \end{enumerate}

  \vspace{0.3cm}
  \textbf{改进方向：}
  \begin{itemize}
    \item 增加噪声强度；
    \item 调整学习率与初始化；
    \item 设计更复杂的非线性模型。
  \end{itemize}

\end{frame}

%--------------------------------------------------
\section{总结与启发}
\begin{frame}{总结与启发}
  \begin{itemize}
    \item 实验验证了梯度下降与拟牛顿法在简单线性模型中的性能；
    \item 手动实现帮助深入理解优化算法的更新机制；
    \item 后续可扩展到非线性或深度模型中，进一步比较不同优化策略的表现。
  \end{itemize}
\end{frame}

\section{改进实验}
\begin{frame}{改进实验}
  \textbf{主要改进点：}
  \begin{itemize}
    \item 使用非线性函数，增加拟合难度；
    \item 统一使用pytorch框架实现GD与BFGS,减小误差；
  \end{itemize}
\end{frame}
